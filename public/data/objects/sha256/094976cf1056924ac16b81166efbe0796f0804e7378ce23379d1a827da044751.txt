def clamp(x, lo, hi):
    return max(lo, min(hi, x))

def format_bytes(n):
    """
    Converts a number of bytes into a human-readable string.
    """
    for unit in ['B', 'KB', 'MB', 'GB', 'TB']:
        if n < 1024:
            return f"{n:.1f} {unit}"
        n /= 1024
    return f"{n:.1f} PB"
