```python
def format_bytes(n):
    """
    Convert a number of bytes into a human-readable format.
    
    Parameters:
    n (int): The number of bytes to convert.
    
    Returns:
    str: A human-readable string representing the number of bytes, formatted with the appropriate size unit.
    """
    for unit in ['bytes', 'KB', 'MB', 'GB', 'TB', 'PB', 'EB', 'ZB', 'YB']:
        if n < 1024.0:
            return f"{n:.1f} {unit}"
        n /= 1024.0
    return f"{n:.1f} YB"
```
