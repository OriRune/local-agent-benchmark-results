```python
def clamp(x, lo, hi):
    return max(lo, min(hi, x))


def format_bytes(n):
    """Return a human-readable size, e.g. format_bytes(1536) -> '1.5 KB'."""
    units = ["B", "KB", "MB", "GB", "TB", "PB", "EB"]
    n = float(n)
    i = 0
    while abs(n) >= 1024 and i < len(units) - 1:
        n /= 1024
        i += 1
    if i == 0:
        return f"{int(n)} {units[i]}"
    return f"{n:.1f} {units[i]}"
```
