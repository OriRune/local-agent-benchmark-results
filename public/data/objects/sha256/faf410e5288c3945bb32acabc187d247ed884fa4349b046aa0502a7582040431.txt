def clamp(x, lo, hi):
    return max(lo, min(hi, x))


def format_bytes(n):
    """Format bytes as a human-readable string (e.g., '1.5 KB')."""
    if n < 1024:
        return f"{n} B"
    elif n < 1024 * 1024:
        return f"{n / 1024:.1f} KB"
    elif n < 1024 * 1024 * 1024:
        return f"{n / (1024 * 1024):.1f} MB"
    else:
        return f"{n / (1024 * 1024 * 1024):.1f} GB"
