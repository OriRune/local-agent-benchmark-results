def clamp(x, lo, hi):
    return max(lo, min(hi, x))


def format_bytes(n):
    """Return a human-readable size, e.g. '1.5 KB'."""
    for unit in ("B", "KB", "MB", "GB", "TB"):
        if abs(n) < 1024:
            return f"{n:.1f} {unit}" if unit != "B" else f"{n:.0f} B"
        n /= 1024
    return f"{n:.1f} PB"
