```python
def format_bytes(n):
    """Return a human-readable byte size, e.g. '1.5 KB'."""
    units = ["B", "KB", "MB", "GB", "TB", "PB"]
    value = float(n)
    i = 0
    while value >= 1024 and i < len(units) - 1:
        value /= 1024
        i += 1
    if i == 0:
        return f"{int(value)} B"
    return f"{value:.1f} {units[i]}"
```
