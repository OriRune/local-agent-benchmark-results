def clamp(x, lo, hi):

def format_bytes(n):
    """Return a human-readable size, e.g. '1.5 KB'."""
    units = ["B", "KB", "MB", "GB", "TB", "PB"]
    value = float(n)
    for unit in units:
        if value < 1024 or unit == units[-1]:
            return f"{value:.1f} {unit}"
        value /= 1024
