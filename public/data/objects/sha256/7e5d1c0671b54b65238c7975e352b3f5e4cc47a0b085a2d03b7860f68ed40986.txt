```python
def format_bytes(n):
    """Return a human-readable size, e.g. '1.5 KB'."""
    units = ["B", "KB", "MB", "GB", "TB", "PB", "EB"]
    value = float(n)
    i = 0
    while abs(value) >= 1024 and i < len(units) - 1:
        value /= 1024
        i += 1
    if i == 0:
        return f"{value:.0f} B"
    return f"{value:.1f} {units[i]}"
```
