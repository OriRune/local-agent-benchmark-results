```python
def clamp(x, lo, hi):
    return max(lo, min(hi, x))

def format_bytes(n):
    units = ['B', 'KB', 'MB', 'GB', 'TB']
    size = n
    for unit in units:
        if size < 1024:
            return f'{size:.1f} {unit}'
        size /= 1024
```
