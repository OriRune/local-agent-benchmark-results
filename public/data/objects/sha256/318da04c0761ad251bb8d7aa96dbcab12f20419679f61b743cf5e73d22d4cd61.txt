def clamp(x, lo, hi):
    return max(lo, min(hi, x))


def format_bytes(n):
    """Return a human-readable byte size string."""
    if n == 0:
        return "0 B"
    units = ["B", "KB", "MB", "GB", "TB", "PB", "EB"]
    for i, unit in enumerate(units):
        if n < 1024**(i+1):
            if unit == "B":
                return f"{n:.0f} B"
            else:
                value = n / (1024**i)
                return f"{value:.1f} {unit}"
    # Fallback for extremely large numbers
    value = n / (1024**6)  # PB
    return f"{value:.1f} PB"
