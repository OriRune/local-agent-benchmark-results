```python
def clamp(x, lo, hi):
    return max(lo, min(hi, x))


def format_bytes(n):
    """Return a human-readable size, e.g. '1.5 KB'."""
    if n < 0:
        raise ValueError("byte size must be non-negative")
    for unit in ("B", "KB", "MB", "GB", "TB"):
        if n < 1024 or unit == "TB":
            if unit == "B":
                return f"{n} {unit}"
            return f"{n:.1f} {unit}"
        n /= 1024
```
